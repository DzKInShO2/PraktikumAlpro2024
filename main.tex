% Numberless chapter command
\newcommand*{\chapterf}[2][]{
    \newpage
    \stepcounter{chapter}
    \chapter*{
        MODUL {\thechapter}\\
        \vspace{1.5ex}
        \MakeUppercase{#2}
    }

    \addcontentsline{toc}{chapter}{MODUL {\thechapter} \MakeUppercase{#2}}
    \thispagestyle{normal}

    \def\a{#1}
    \ifx\a\empty
        \renewcommand*{\leftmark}{#2}
    \else
        \renewcommand*{\leftmark}{#1}
    \fi
}
