% Numberless chapter command
\newcommand*{\chapterf}[2][]{
    \def\a{#1}
    \def\e{}
    \stepcounter{chapter}
    \chapter*{
        MODUL {\thechapter}\\\vspace{1.5ex}\MakeUppercase{#2}
    }
    \ifx\a\e
        \markboth{#2}{}
    \else
        \markboth{#1}{}
    \fi
    \addcontentsline{toc}{chapter}{MODUL {\thechapter} #2}
}

% import figure
\newcommand*{\inputfigure}[5][]{
    \def\a{#1}
    \def\n{n}
    \def\p{p}
    \begin{figure}[H]
        \centering
        \setstretch{1.0}
        \renewcommand*{\thechapter}{\arabic{chapter}}

        \ifx\a\n
            \includegraphics[
                scale=#3,
                center,
                frame=0.5pt
            ]{#2}
        \else
            \ifx\a\p
                \begin{indentlevel0}
                    \includegraphics[
                        scale=#3,
                        center,
                        frame=0.5pt
                    ]{#2}
                \end{indentlevel0}
            \fi
        \fi

        \caption{#4}
        \label{#5}
        \vspace*{-2.0ex}
    \end{figure}
}

% create table
\newcommand*{\inputtable}[5][]{
    \vspace{-1ex}
	\cpsize
    \renewcommand*{\thechapter}{\arabic{chapter}}
	\ifthenelse{\isempty{#1}}{
		\begin{xltabular}{\textwidth}{#2}
	}{
		\ifthenelse{\equal{#1}{n}}{
			\begin{xltabular}[r]{0.95\textwidth}{#2}
		}{
            \begin{xltabular}[r]{#1\textwidth}{#2}
		}
	}
        \caption{#4}
        \label{#5}
        \\ \hline
        #3
    \end{xltabular}
	\normalsize
	\vspace*{-12pt}
}
